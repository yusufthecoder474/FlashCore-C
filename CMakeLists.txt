cmake_minimum_required(VERSION 3.16)
project(FlashCore_C LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable ASan and UBSan on GCC/Clang" OFF)
option(ENABLE_ANALYZER "Enable GCC static analyzer" OFF)
if(MSVC)
  message(FATAL_ERROR "Strict C99 build uses GCC or Clang. On Windows use MSYS2 UCRT64 GCC.")
endif()
add_compile_options(-Wall -Wextra -Werror -pedantic -Wvla)
if(ENABLE_SANITIZERS)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" OR WIN32)
    message(FATAL_ERROR "This sanitizer configuration supports GCC/Clang on Linux; Windows pending.")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
if(ENABLE_ANALYZER)
  if(NOT CMAKE_C_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "ENABLE_ANALYZER requires GCC")
  endif()
  add_compile_options(-fanalyzer)
endif()
add_library(ftl src/ftl.c)
target_include_directories(ftl PUBLIC include)
add_executable(flashcore src/cli.c)
target_link_libraries(flashcore PRIVATE ftl)
add_executable(test_ftl tests/test_ftl.c)
target_link_libraries(test_ftl PRIVATE ftl)
enable_testing()
foreach(case initial bounds overwrite trim program_fault delayed_fault gc_copy gc_rollback gc_fault_positions erase_fault double_fault retirement wear_choice exclude_active exhausted corruption stress)
  add_test(NAME ${case} COMMAND test_ftl ${case})
endforeach()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME python_validation COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_CURRENT_SOURCE_DIR}/scripts/validate.py --exe $<TARGET_FILE:flashcore>
  --out ${CMAKE_CURRENT_BINARY_DIR}/validation)

add_test(NAME comparison_tests COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_CURRENT_SOURCE_DIR}/scripts/test_compare.py)
